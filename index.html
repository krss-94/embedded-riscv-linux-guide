    Embedded RISC-V Linux — Step by Step Command Guide

    Consolidated from class notes 07/06 – 07/17. Every command line shows the prompt (user@host:path) it was run from.
    Mirror IP note: school notes use 10.0.1.147 or 10.0.1.148 (campus-only repo). Off-campus, swap that IP for deb.debian.org wherever it appears.

    STEP 3FIRST LOGIN & BASIC LINUX COMMANDS
    # Log in
    username: hunter
    password: abc123
    # Check who you are
    hunter@debian:~$ whoami
    # Become root
    hunter@debian:~$ su -
    # password: abc123
    root@debian:~# whoami
    # Check Debian version
    hunter@debian:~$ cat /etc/os-release
    hunter@debian:~$ cat /etc/debian_version
    # Basic navigation commands
    hunter@debian:~$ ls
    hunter@debian:~$ ls -l
    hunter@debian:~$ cd /etc
    hunter@debian:/etc$ cd ..
    hunter@debian:~$ cd .
    hunter@debian:~$ cd
    hunter@debian:~$ pwd
    hunter@debian:~$ mkdir prog
    hunter@debian:~$ rm -rf prog
    # Shutdown the machine
    hunter@debian:~$ su -
    root@debian:~# shutdown -h now
    STEP 4UPDATE / UPGRADE SYSTEM (root, do weekly)
    root@debian:~# apt-get update
    root@debian:~# apt-get upgrade
    # NOTE: never run apt-get in two terminals at the same time
    # (causes "Could not get lock /var/lib/dpkg/lock-frontend" error)
    STEP 5TEST PACKAGE INSTALL/REMOVE (optional demo)
    root@debian:~# apt-get install x11-apps xorg
    hunter@debian:~$ xeyes
    # Ctrl+C to quit
    root@debian:~# apt-get remove x11-apps
    # or, alternative demo package used in a later session:
    root@debian:~# apt-get install xosview
    hunter@debian:~$ xosview
    # Ctrl+C to quit
    root@debian:~# apt-get remove xosview
    STEP 6CHECK SYSTEM STATUS / INFO
    hunter@debian:~$ cat /proc/cpuinfo
    hunter@debian:~$ df -h
    hunter@debian:~$ lsblk
    hunter@debian:~$ free -h
    hunter@debian:~$ lspci
    STEP 7INSTALL X86_64 TOOLCHAIN
    hunter@debian:~$ gcc -v
    hunter@debian:~$ g++ -v
    hunter@debian:~$ cpp -v
    root@debian:~# apt-get update
    root@debian:~# apt-get install gcc cpp g++
    # verify
    hunter@debian:~$ gcc -v
    hunter@debian:~$ g++ -v
    hunter@debian:~$ cpp -v
    STEP 8INSTALL RISC-V CROSS TOOLCHAIN
    root@debian:~# apt-get install crossbuild-essential-riscv64 \
    gcc-riscv64-linux-gnu \
    cpp-riscv64-linux-gnu \
    g++-riscv64-linux-gnu \
    u-boot-qemu u-boot-tools opensbi \
    gcc g++ cpp \
    m4 make nasm bison flex autoconf \
    automake autotools-dev libncurses5-dev \
    dpkg-dev libssl-dev uuid-dev \
    libgnutls28-dev libelf-dev \
    rsync clang llvm lld bc
    # verify
    hunter@debian:~$ riscv64-linux-gnu-gcc -v
    hunter@debian:~$ riscv64-linux-gnu-g++ -v
    hunter@debian:~$ riscv64-linux-gnu-cpp -v
    STEP 9WRITE & CROSS-COMPILE A TEST C PROGRAM
    hunter@debian:~$ mkdir prog
    hunter@debian:~$ cd prog
    hunter@debian:~/prog$ nano hello.c
    # type your C code, Ctrl+O to save, Ctrl+X to exit
    hunter@debian:~/prog$ cat hello.c
    # compile for host (x86_64)
    hunter@debian:~/prog$ gcc hello.c -o hello_x86-64
    hunter@debian:~/prog$ ./hello_x86-64
    # cross-compile for RISC-V
    hunter@debian:~/prog$ riscv64-linux-gnu-gcc hello.c -o hello_riscv64
    hunter@debian:~/prog$ ./hello_riscv64
    # will fail here: Exec format error (expected — wrong CPU arch)
    # inspect binaries
    hunter@debian:~/prog$ file hello.c
    hunter@debian:~/prog$ file hello_x86-64
    hunter@debian:~/prog$ file hello_riscv64
    STEP 10SET UP BRIDGE INTERFACE (br10)
    root@debian:~# apt-get install bridge-utils uml-utilities
    root@debian:~# cd /etc
    root@debian:/etc# mkdir qemu
    root@debian:/etc# cd qemu
    root@debian:/etc/qemu# nano bridge.conf
    # content:
    allow br10
    root@debian:/etc/qemu# nano /etc/network/interfaces
    # add:
    auto br10
    iface br10 inet static
    address 192.168.10.1
    netmask 255.255.255.0
    #gateway 10.0.2.15
    bridge_ports tap10
    bridge_fd 9
    bridge_hello 2
    bridge_maxage 12
    bridge_stp off
    # comment out (or omit) the "gateway" line to avoid the
    # duplicate default-route problem described in Step 10b.
    root@debian:/etc/qemu# reboot
    # verify after reboot
    hunter@debian:~$ ip addr show
    hunter@debian:~$ ping 192.168.10.1
    # Ctrl+C to stop ping
    STEP 10bFIX DUPLICATE DEFAULT ROUTE (if seen)
    hunter@debian:~$ ip route
    # if you see two "default via ..." lines (one via br10):
    root@debian:~# ip route del default dev br10
    root@debian:~# ip route
    # permanent fix: edit /etc/network/interfaces and comment out the
    # "gateway 10.0.2.15" line under br10, then:
    root@debian:~# /etc/init.d/networking restart
    root@debian:~# ip route
    STEP 11INSTALL QEMU
    root@debian:~# apt-get install qemu-user-static qemu-system-riscv64
    root@debian:~# chmod u+s /usr/lib/qemu/qemu-bridge-helper
    STEP 12DOWNLOAD CLASS BUILD SCRIPTS
    hunter@debian:~$ wget http://10.0.1.147/virtualriscv.tar.xz
    hunter@debian:~$ tar xfva virtualriscv.tar.xz
    hunter@debian:~$ cd ~/virtualriscv
    STEP 13CONFIGURE & COMPILE LINUX KERNEL (6.1.177)
    hunter@debian:~/virtualriscv$ bash build_kernel.sh
    STEP 14CREATE ROOT FILESYSTEM (debootstrap)
    root@debian:~# apt-get install debootstrap
    # edit mirror IP in the script first if needed:
    hunter@debian:~/virtualriscv$ nano debootstrap_rsicv.sh
    # change: DEBIAN_MIRROR=http://<old-ip>//$DEBIAN_VERSION
    # to: DEBIAN_MIRROR=http://10.0.1.147//$DEBIAN_VERSION
    # (or http://deb.debian.org//debian if off-campus)
    root@debian:~# cd /home/hunter/virtualriscv
    root@debian:/home/hunter/virtualriscv# bash debootstrap_rsicv.sh
    # if download fails, re-fetch script and retry:
    hunter@debian:~/virtualriscv$ rm -f debootstrap_rsicv.sh
    hunter@debian:~/virtualriscv$ wget http://10.0.1.147/debootstrap_riscv.sh
    root@debian:/home/hunter/virtualriscv# bash debootstrap_rsicv.sh
    # rootfs defaults created by the script:
    # web page: /var/www/html/index.html -> "Welcome to RISC-V LeeNIX!!"
    # view at http://192.168.10.100:8000
    # default account created inside rootfs: leenix / leenix
    # (root password inside rootfs set to: leenix)
    STEP 15COMPILE OPENSBI
    hunter@debian:~/virtualriscv$ bash build_opensbi.sh
    STEP 16COMPILE U-BOOT (2024.10)
    hunter@debian:~/virtualriscv$ bash build_uboot.sh
    STEP 17(OPTIONAL) FIX/UPDATE APT MIRROR ON THE HOST VM
    root@debian:~# cd /etc/apt
    root@debian:/etc/apt# nano sources.list
    # On campus, lines should point to:
    # deb http://10.0.1.147/debian/ trixie main contrib non-free non-free-firmware
    # deb http://security.debian.org/debian-security trixie-security main non-free-firmware
    # deb http://10.0.1.147/debian/ trixie-updates main contrib non-free non-free-firmware
    #
    # Off campus, replace 10.0.1.147 with deb.debian.org, e.g.:
    # deb http://deb.debian.org/debian trixie main contrib non-free non-free-firmware
    root@debian:/etc/apt# apt-get update
    root@debian:/etc/apt# apt-get upgrade
    # alternative: fetch a ready-made sources.list from the class server
    root@debian:/etc/apt# rm -f sources.list
    root@debian:/etc/apt# wget http://10.0.1.147/sources.list
    root@debian:/etc/apt# apt-get update
    STEP 18CREATE THE VIRTUAL SD CARD IMAGE
    hunter@debian:~/virtualriscv$ cd ~/virtualriscv
    root@debian:/home/hunter/virtualriscv# bash createsdimg.sh
    # inspect the resulting image
    root@debian:/home/hunter/virtualriscv# ls -l sd_1G.img
    root@debian:/home/hunter/virtualriscv# file sd_1G.img
    root@debian:/home/hunter/virtualriscv# fdisk -lu sd_1G.img
    # if a revised script is needed:
    root@debian:/home/hunter/virtualriscv# rm createsdimg.sh
    root@debian:/home/hunter/virtualriscv# wget http://10.0.1.147/createsdimg.sh
    STEP 19COPY KERNEL + ROOTFS + U-BOOT ONTO THE SD IMAGE
    root@debian:/home/hunter/virtualriscv# df -h
    # check space before copying
    root@debian:/home/hunter/virtualriscv# bash cprootfs.sh
    # copy rootfs onto sd image
    root@debian:/home/hunter/virtualriscv# df -h
    root@debian:/home/hunter/virtualriscv# bash cpsdcard.sh
    # copy kernel/u-boot onto sd image
    root@debian:/home/hunter/virtualriscv# df -h
    root@debian:/home/hunter/virtualriscv# bash unsetsdcard.sh
    # unmount sd image / clear loop device
    root@debian:/home/hunter/virtualriscv# df -h
    # fix ownership (script runs as root, so the image ends up root:root)
    root@debian:/home/hunter/virtualriscv# ls -l sd_1G.img
    root@debian:/home/hunter/virtualriscv# chown hunter:hunter sd_1G.img
    root@debian:/home/hunter/virtualriscv# ls -l sd_1G.img
    STEP 20BOOT THE RISC-V SYSTEM IN QEMU
    hunter@debian:~/virtualriscv$ bash runsd.sh
    # login once it boots to the prompt (riscvleenix login:)
    riscvleenix login: root
    Password: leenix
    # (rootfs also has a "leenix" user with the same password: leenix)
    STEP 21VERIFY THE RUNNING RISC-V SYSTEM
    root@riscvleenix:~# cat /proc/cpuinfo
    # confirm rv64 architecture
    root@riscvleenix:~# uname -a
    # confirm kernel 6.1.177 riscv64
    root@riscvleenix:~# df -h
    # confirm /dev/root usage < 256 MB
    root@riscvleenix:~# free -h
    # confirm ~1.8-2 GB memory available
